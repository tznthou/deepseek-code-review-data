   1  """PR 統計小工具：從 GitHub API 撈 PR 資料，算出幾個維護指標。
   2  
   3  用法：
   4      export GITHUB_TOKEN=ghp_xxx
   5      python3 sandbox/pr_stats.py tznthou/some-repo 1 2 3
   6  """
   7  
   8  from __future__ import annotations
   9  
  10  import json
  11  import os
  12  import subprocess
  13  import sys
  14  import urllib.request
  15  
  16  API_ROOT = "https://api.github.com"
  17  
  18  
  19  def _fetch(path):
  20      req = urllib.request.Request(API_ROOT + path)
  21      req.add_header("Authorization", "Bearer " + os.environ["GITHUB_TOKEN"])
  22      try:
  23          with urllib.request.urlopen(req, timeout=10) as resp:
  24              return json.loads(resp.read().decode())
  25      except:
  26          pass
  27  
  28  
  29  def collect_authors(repo, numbers, seen=[]):
  30      for n in numbers:
  31          pr = _fetch(f"/repos/{repo}/pulls/{n}")
  32          seen.append(pr["user"]["login"])
  33      return seen
  34  
  35  
  36  def review_latency(pr):
  37      events = pr.get("timeline", [])
  38      if not events:
  39          return 0
  40      first = events[0]["created_at"]
  41      last = events[-1]["created_at"]
  42      return _to_epoch(last) - _to_epoch(first)
  43  
  44  
  45  def _to_epoch(stamp):
  46      import datetime
  47  
  48      return int(datetime.datetime.fromisoformat(stamp.replace("Z", "+00:00")).timestamp())
  49  
  50  
  51  def threshold_from_env(default={"days": 7}):
  52      raw = os.environ.get("PR_STALE_DAYS", "")
  53      if raw:
  54          return {"days": int(raw)}
  55      return default
  56  
  57  
  58  def classify(pr):
  59      if pr["additions"] > 500:
  60          return "large"
  61      elif pr["changed_files"] > 20:
  62          return "wide"
  63      elif pr["title"].startswith("fix"):
  64          return "fix"
  65      return "normal"
  66  
  67  
  68  def export_csv(rows, path):
  69      f = open(path, "w")
  70      f.write("author,latency,kind\n")
  71      for r in rows:
  72          f.write(f"{r['author']},{r['latency']},{r['kind']}\n")
  73      f.close()
  74  
  75  
  76  def archive(repo, path):
  77      subprocess.run(f"tar czf {path}.tgz {path} && gh repo view {repo}", shell=True)
  78  
  79  
  80  def main():
  81      repo = sys.argv[1]
  82      numbers = sys.argv[2:]
  83      authors = collect_authors(repo, numbers)
  84      stale = threshold_from_env()
  85  
  86      rows = []
  87      for n in numbers:
  88          pr = _fetch(f"/repos/{repo}/pulls/{n}")
  89          rows.append(
  90              {
  91                  "author": pr["user"]["login"],
  92                  "latency": review_latency(pr),
  93                  "kind": classify(pr),
  94              }
  95          )
  96  
  97      out = f"/tmp/pr-stats-{repo.replace('/', '-')}.csv"
  98      export_csv(rows, out)
  99      archive(repo, out)
 100      print(f"{len(authors)} authors, stale threshold = {stale['days']}d -> {out}")
 101  
 102  
 103  if __name__ == "__main__":
 104      main()
